from urllib.parse import urlparse


def get_domain(url):
    """Get the domain name from a URL."""
    try:
        domain = urlparse(url).hostname

        if domain:
            return domain.lower()

        return ""

    except ValueError:
        return ""


def analyze_email(sender, subject, message,
                  displayed_url, actual_url, final_url):

    risk_score = 0
    warnings = []

    # --------------------------------
    # CHECK 1: SENDER
    # --------------------------------

    if "@" not in sender:
        risk_score += 20
        warnings.append(
            "Sender email address looks suspicious."
        )

    # --------------------------------
    # CHECK 2: EMAIL CONTENT
    # --------------------------------

    suspicious_words = [
        "urgent",
        "immediately",
        "verify your account",
        "password",
        "click here",
        "account suspended"
    ]

    email_text = (subject + " " + message).lower()

    for word in suspicious_words:
        if word in email_text:
            risk_score += 10
            warnings.append(
                "Suspicious phrase found: " + word
            )

    # --------------------------------
    # CHECK 3: HTTP VS HTTPS
    # --------------------------------

    if actual_url.startswith("http://"):
        risk_score += 20
        warnings.append(
            "The actual link does not use HTTPS."
        )

    # --------------------------------
    # GET THE DOMAINS
    # --------------------------------

    displayed_domain = get_domain(displayed_url)
    actual_domain = get_domain(actual_url)
    final_domain = get_domain(final_url)

    # --------------------------------
    # CHECK 4:
    # DISPLAYED LINK VS ACTUAL LINK
    # --------------------------------

    if displayed_domain and actual_domain:

        if displayed_domain != actual_domain:
            risk_score += 30

            warnings.append(
                "Displayed link does not match "
                "the actual hyperlink."
            )

    # --------------------------------
    # CHECK 5:
    # ACTUAL LINK VS FINAL DESTINATION
    # --------------------------------

    if actual_domain and final_domain:

        if actual_domain != final_domain:
            risk_score += 30

            warnings.append(
                "The link redirects to a "
                "different domain."
            )

    # --------------------------------
    # DETERMINE RISK LEVEL
    # --------------------------------

    if risk_score >= 50:
        risk_level = "HIGH RISK"

    elif risk_score >= 20:
        risk_level = "SUSPICIOUS"

    else:
        risk_level = "LEGITIMATE"

    # --------------------------------
    # RETURN RESULTS
    # --------------------------------

    return {
        "risk_score": risk_score,
        "risk_level": risk_level,
        "displayed_domain": displayed_domain,
        "actual_domain": actual_domain,
        "final_domain": final_domain,
        "warnings": warnings
    }


# ==================================

    sender="security@example.com",

    subject="URGENT - Verify Your Account",

    message=(
        "Your account has been suspended. "

    # What the person sees

    # Link stored inside the email
    actual_url="https://example.com",

    # Where the link eventually sends the person
    final_url="https://different-example.com"
)
